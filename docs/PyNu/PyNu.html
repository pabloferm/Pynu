174    def ApplyWeights(self, tag, vector=None):
175        if tag == 'Fixed':
176            labels = self.Analysis.Fixed
177            vec = self.Analysis.FixedValue
178        elif tag == 'Nominal':
179            labels = self.Analysis.Nuisance
180            vec = self.Analysis.NuisNominal
181        elif tag == 'True':
182            labels = self.Analysis.Physics
183            vec = self.Analysis.PhysTrue
184        elif tag == 'Physics':
185            labels = self.Analysis.Physics
186            v_id = self.Analysis.PhysicsList
187        elif tag == 'Nuisance':
188            labels = self.Analysis.Nuisance
189            v_id = self.Analysis.NuisanceList
190        else:
191            sys.exit('Not a valid tag for applying weights.')
192
193        for name, exp in self.Experiments.items():
194            for source in labels:
195                if source in exp.Definition.keys():
196                    tune_block = exp.Definition[source]
197                    for tune in labels[source]:
198                        if vector is not None:
199                            idx = v_id.index(tune)
200                            value = vector[idx]
201                        else:
202                            value = vec[source][tune]
203
204                        if tune_block == 'Flux':
205                            w = self.PhysicsTunes[name].GetFlux(tune, value)
206                        elif tune_block == 'XSection':
207                            w = self.PhysicsTunes[name].GetXSection(
208                                tune, value)
209                        elif tune_block == 'Detector':
210                            w = self.PhysicsTunes[name].GetDetector(
211                                tune, value)
212                        elif tune_block == 'Osc':
213                            self.PhysicsTunes[name].OscillationTunes.UpdateParameter(
214                                tune, value)
215
216                        if tune_block != 'Osc':
217                            if tag == 'Fixed':
218                                exp.UpdateBaseWeights(w)
219                            elif tag in ['True', 'Nominal']:
220                                exp.UpdateNominalWeights(w)
221                            elif tag == 'Physics':
222                                exp.UpdatePhysicsWeights(w)
223                            elif tag == 'Nuisance':
224                                exp.UpdateNuisanceWeights(w)

226    def GetDiffLogWeights(self, vector):
227        ''' Computes the derivative with respect the nuisance parameter nuis '''
228        ''' Returns a dict of nuis : experiment : partial of weight with respect to nuis over weight '''
229        dWoverW = {}
230
231        for source, nuisance_list in self.Analysis.Nuisance.items():
232            for name, exp in self.Experiments.items():
233                if source in exp.Definition.keys():
234                    tune_block = exp.Definition[source]
235                    for tune in self.Analysis.Nuisance[source]:
236                        dWoverW[tune] = {name: 0}
237                        idx = self.Analysis.NuisanceList.index(tune)
238                        if tune_block == 'Flux':
239                            dWoverW[tune][name] = self.PhysicsTunes[name].GetFlux(
240                                'Diff_' + tune, vector[idx]) / self.PhysicsTunes[name].GetFlux(tune, vector[idx])
241                        elif tune_block == 'XSection':
242                            dWoverW[tune][name] = self.PhysicsTunes[name].GetXSection(
243                                'Diff_' + tune, vector[idx]) / self.PhysicsTunes[name].GetXSection(tune, vector[idx])
244                        elif tune_block == 'Detector':
245                            dWoverW[tune][name] = self.PhysicsTunes[name].GetDetector(
246                                'Diff_' + tune, vector[idx]) / self.PhysicsTunes[name].GetDetector(tune, vector[idx])
247                        elif tune_block == 'Osc':
248                            dWoverW[tune][name] = self.PhysicsTunes[name].GetOscillation(
249                                'Diff_' + tune, vector[idx]) / self.PhysicsTunes[name].OscillationTunes.GetOscillations()
250        return dWoverW

252    def FitModel(self, point, mode='BinnedLogLikelihoodRatio'):
253        if mode == 'BinnedLogLikelihoodRatio':
254            from scipy.optimize import minimize
255
256            ''' Binned log-Likelihood fit assuming data is Poisson-distributed '''
257            self.LLH = FT.BinnedLogLikelihoodRatio(
258                self.Observation,
259                self.Analysis.NuisNominalList,
260                self.Analysis.NuisSigmaList,
261                self.Analysis.NuisDistributionList)
262
263            ''' Binned log-Likelihood fit assuming data is Poisson-distributed '''
264            self.ComputeBinnedExpectation(
265                point, physics=True)  # Nominal expectation
266            ''' Statistics only computation to start guiding the minimization '''
267            X2_stats = self.LLH.StatsOnly(self.Expectation)
268
269            '''Get Jacobian of expected events w.r.t. nuisance parameters'''
270            self.ComputeBinnedDiffExpectation()
271
272            '''Analytic estimate for priors and bounds at first order'''
273            AnalyticPrior, AnalyticBounds = self.LLH.AnalyticPriorsBounds(
274                self.Expectation, self.DiffExpectation)
275
276            '''Combined chi^2 minimization'''
277            if X2_stats > 200:
278                tol = 1e-4
279            else:
280                tol = None
281            res = minimize(
282                self.ModelTester,
283                AnalyticPrior,
284                args=(point),
285                # method='L-BFGS-B',
286                jac=True,
287                bounds=AnalyticBounds,
288                options={
289                    'disp': False,
290                    'tol': tol})
291
292            self.WriteToOutFile(
293                point,
294                'Analysis',
295                'Chi2 Stats. Only',
296                X2_stats)
297
298            self.WriteToOutFile(
299                point,
300                'Nuisance Parameters',
301                self.Analysis.NuisanceList,
302                res.x.tolist())
303
304            self.WriteToOutFile(point, 'Analysis', 'Chi2 Systs.', res.fun)
305
306            if self.verbosity:
307                print(f'Fitted nuisances: {res.x}')
308                print(f'-2 ln(H/H0) = {res.fun}')
309
310            return - 0.5 * res.fun

312    def ModelTester(self, nuisance_vector, point):
313        ''' Compute expected and its derivatives '''
314        self.ComputeBinnedExpectation(
315            point, nuisance_vector=nuisance_vector)  # Nominal expectation
316        self.ComputeBinnedDiffExpectation(nuisance_vector=nuisance_vector)
317
318        ''' Get -2 ln(H/H0) ~ χ2 '''
319        Chi2 = self.LLH.StatsAndSystematics(
320            self.Expectation,
321            nuisance_vector)
322
323        ''' The gradient of the above '''
324        D_Chi2 = self.LLH.Gradient(
325            self.Expectation,
326            self.DiffExpectation,
327            nuisance_vector)
328
329        return (Chi2, D_Chi2)

334    def CreateOutFile(self, fname):
335        import h5py
336        self.outfile = fname
337        with h5py.File(fname, 'w') as hf:
338            grp = hf.create_group('Fixed Parameters')
339            for key in self.Analysis.Fixed.keys():
340                this = grp.create_group(key)
341                for par, val in self.Analysis.FixedValue[key].items():
342                    this.create_dataset(par, data=[val], compression='gzip')
343
344            if self.Analysis.wSyst:
345                grp = hf.create_group('Nuisance Parameters')
346                for key in self.Analysis.Nuisance.keys():
347                    this = grp.create_group(key)
348                    for par in self.Analysis.Nuisance[key]:
349                        this.create_dataset(
350                            par,
351                            data=[0.0] *
352                            self.Analysis.NumberOfPhysPoints,
353                            compression='gzip')
354            grp = hf.create_group('Physics Parameters')
355
356            physics_lists = list(zip(*self.Analysis.FullPhysicsGrid))
357            for key in self.Analysis.Physics.keys():
358                this = grp.create_group(key)
359                for par in self.Analysis.Physics[key]:
360                    idx = self.Analysis.PhysicsList.index(par)
361                    this.create_dataset(
362                        par, data=physics_lists[idx],
363                        compression='gzip')
364
365            grp = hf.create_group('Analysis')
366            grp.create_dataset(
367                'Chi2 Stats. Only',
368                data=[0.0] *
369                self.Analysis.NumberOfPhysPoints,
370                compression='gzip')
371            if self.Analysis.wSyst:
372                grp.create_dataset(
373                    'Chi2 Systs.',
374                    data=[0.0] *
375                    self.Analysis.NumberOfPhysPoints,
376                    compression='gzip')

378    def WriteToOutFile(self, point, block, item, value):
379        import h5py
380        with h5py.File(self.outfile, 'r+') as hf:
381            try:
382                for par, val in zip(item, value):
383                    source = self.Analysis.GetSourceOfTune(par)
384                    hf[block + '/' + source + '/' + par][point] = val
385            except BaseException:
386                hf[block + '/' + item][point] = value
